//   1) tam migrate edilmis (001-030) BOS bir PostgreSQL verin (asagidaki DATABASE_URL satirini kendi hedefinize cevirin; UYGULAMA/URETIM DB OLMASIN)
//   2) cd server && node ../docs/superpowers/analysis/wp-h-smoke-e2e.js.txt   (cwd = server/, `.env` OLMAMALI; MQTT yok -> kopru baslamaz)
// Yaptigi: gercek sunucuyu start() ile kaldirir, gercek girisle belirtec alir, push-token uclarini, peace GET v2 ve close-all (502 yolu) uclarini sinar, zarif kapanisi dogrular, kendi satirlarini siler.
// Entegre kopyada (yama uygulanmis) GERCEK sunucu + GERCEK PostgreSQL ile uctan uca duman testi.
// Calistirma: cwd = integ/server. MQTT yok (kopru baslamaz -> komutlar 502). Gizli deger yok (rastgele, bu surece ozel).
const path = require('path');
const crypto = require('crypto');
const SERVER = process.cwd();
const rnd = (n) => crypto.randomBytes(n).toString('hex');

process.env.NODE_ENV = 'test';
process.env.DATABASE_URL = 'postgresql://postgres@127.0.0.1:55432/ev_peace_test';
process.env.JWT_SECRET = rnd(24);
process.env.PIN_PEPPER = rnd(24);
process.env.LOCAL_KEY_SECRET = rnd(32);
process.env.PORT = '55080';
process.env.BIND_HOST = '127.0.0.1';
process.env.MQTT_HOST = '127.0.0.1';
process.env.MQTT_PORT = '55883';
process.env.SMTP_HOST = '127.0.0.1';
delete process.env.MQTT_BACKEND_USER;
delete process.env.MQTT_BACKEND_PASS;
delete process.env.PEACE_REMINDER_ENABLED;
process.env.PEACE_REMINDER_DRY_RUN = 'true';

const errors = [];
const logs = [];
const origErr = console.error;
const origLog = console.log;
const origWarn = console.warn;
console.error = (...a) => { errors.push(a.map(String).join(' ')); };
console.log = (...a) => { logs.push(a.map(String).join(' ')); };
console.warn = (...a) => { logs.push('WARN ' + a.map(String).join(' ')); };
let exitCode = null;
const realExit = process.exit;
process.exit = (c) => { exitCode = c; };

const { Pool } = require(path.join(SERVER, 'node_modules/pg'));
const bcrypt = require(path.join(SERVER, 'node_modules/bcryptjs'));
const assert = require('assert');

(async () => {
  const pool = new Pool({ connectionString: process.env.DATABASE_URL, max: 4 });
  const q = (t, p) => pool.query(t, p);
  const tag = rnd(4);
  const email = `smoke_${tag}@example.invalid`;
  const password = 'Smoke-' + rnd(6);
  let userId; let homeId; let devId;
  const results = [];
  const ok = (name) => results.push('OK   ' + name);
  const step = async (name, fn) => { try { await fn(); ok(name); } catch (e) { results.push('FAIL ' + name + ' :: ' + (e && e.message)); } };

  try {
    // ---- fixture
    userId = (await q("INSERT INTO users (email, password_hash, full_name, role, is_active) VALUES ($1, $2, 'Duman Test', 'user', TRUE) RETURNING id", [email, await bcrypt.hash(password, 4)])).rows[0].id;
    await q("UPDATE users SET email_verified = TRUE, account_status = 'active' WHERE id = $1", [userId]).catch(() => {});
    homeId = (await q("INSERT INTO homes (name, mqtt_username, timezone) VALUES ($1, $2, 'Europe/Istanbul') RETURNING id", [`Duman-${tag}`, `h_${rnd(8)}`])).rows[0].id;
    await q("INSERT INTO home_users (home_id, user_id, role) VALUES ($1, $2, 'owner')", [homeId, userId]);
    devId = (await q('INSERT INTO devices (home_id, device_uuid, mac_address, is_claimed, is_online, last_seen_at) VALUES ($1, $2, $3, TRUE, TRUE, now()) RETURNING id', [homeId, `AHBU-SMK${tag}`.toUpperCase(), `SMK${tag}`])).rows[0].id;
    const eps = [[1, 'shutter', 'Salon', 100], [2, 'shutter', 'Salon', 100], [3, 'light', 'Salon', 0], [4, 'light', 'Mutfak', 0]];
    for (const [ch, type, room, pos] of eps) {
      await q('INSERT INTO endpoints (home_id, device_id, channel_index, name, type, room, current_state, current_position) VALUES ($1,$2,$3,$4,$5,$6,$7,$8)', [homeId, devId, ch, `K${ch}`, type, room, type === 'light', pos]);
    }
    const noticeId = (await q("INSERT INTO peace_notification_logs (home_id, local_date, status, open_lights_count, open_shutters_count, summary_text) VALUES ($1, CURRENT_DATE, 'sent', 2, 1, 'Salonda 1, Mutfakta 1 lamba açık.') RETURNING id", [homeId])).rows[0].id;

    // ---- sunucu
    const { start } = require(path.join(SERVER, 'src/server.js'));
    const { app, server, shutdown } = start();
    await new Promise((r) => (server.listening ? r() : server.once('listening', r)));
    const base = 'http://127.0.0.1:55080';
    const call = async (method, url, { token, body } = {}) => {
      const res = await fetch(base + url, { method, headers: { 'content-type': 'application/json', ...(token ? { authorization: 'Bearer ' + token } : {}) }, body: body === undefined ? undefined : JSON.stringify(body) });
      let json = null; try { json = await res.json(); } catch (_) { /* gövde yok */ }
      return { status: res.status, json, headers: res.headers };
    };

    await step('sunucu ayakta; pushService app.locals icinde', async () => {
      assert.strictEqual((await call('GET', '/health')).status, 200);
      assert.ok(app.locals.pushService && typeof app.locals.pushService.upsertToken === 'function');
    });
    await step('/ready: veritabani up', async () => {
      const r = await call('GET', '/ready');
      assert.strictEqual(r.json.components.database, 'up');
    });
    await step('kimliksiz PUT /me/push-tokens -> 401', async () => {
      const r = await call('PUT', '/api/v1/me/push-tokens', { body: { token: 'x'.repeat(30), platform: 'android' } });
      assert.strictEqual(r.status, 401, JSON.stringify(r.json));
    });
    let token;
    await step('login -> access token', async () => {
      const r = await call('POST', '/api/v1/auth/login', { body: { email, password } });
      assert.strictEqual(r.status, 200, JSON.stringify(r.json));
      token = r.json && r.json.data && (r.json.data.access_token || r.json.data.accessToken || r.json.data.token);
      assert.ok(token, 'token yok: ' + JSON.stringify(Object.keys(r.json.data || {})));
    });
    const fcm = 'smoke-token-' + rnd(16);
    await step('PUT /api/v1/me/push-tokens -> 200, satir kullaniciya baglandi', async () => {
      const r = await call('PUT', '/api/v1/me/push-tokens', { token, body: { token: fcm, platform: 'android', app_version: '1.0.0' } });
      assert.strictEqual(r.status, 200, JSON.stringify(r.json));
      assert.strictEqual(r.json.data.registered, true);
      assert.strictEqual(r.headers.get('cache-control'), 'no-store');
      const row = (await q('SELECT user_id, platform, disabled_at FROM push_tokens WHERE token = $1', [fcm])).rows[0];
      assert.ok(row && row.user_id === userId && row.platform === 'android' && row.disabled_at === null);
    });
    await step('eski /api/me/push-tokens baglantisi da calisir', async () => {
      const r = await call('PUT', '/api/me/push-tokens', { token, body: { token: fcm, platform: 'android' } });
      assert.strictEqual(r.status, 200, JSON.stringify(r.json));
    });
    await step('gecersiz govde -> 400 VALIDATION (token cok kisa / platform gecersiz)', async () => {
      assert.strictEqual((await call('PUT', '/api/v1/me/push-tokens', { token, body: { token: 'kisa', platform: 'android' } })).json.code, 'VALIDATION');
      assert.strictEqual((await call('PUT', '/api/v1/me/push-tokens', { token, body: { token: fcm, platform: 'windows' } })).json.code, 'VALIDATION');
    });
    await step('GET /api/v1/devices/peace-notification/:home -> v2 yanit, no-store', async () => {
      const r = await call('GET', `/api/v1/devices/peace-notification/${homeId}`, { token });
      assert.strictEqual(r.status, 200, JSON.stringify(r.json));
      const d = r.json.data;
      assert.strictEqual(d.open_lights_count, 2);
      assert.strictEqual(d.open_shutters_count, 1);
      assert.strictEqual(d.stale, false);
      assert.strictEqual(d.timezone, 'Europe/Istanbul');
      assert.strictEqual(d.last_notice && d.last_notice.id, noticeId);
      assert.strictEqual(r.headers.get('cache-control'), 'no-store');
    });
    await step('POST close-all (MQTT yok) -> 502 BROKER_UNAVAILABLE; bildirim COZULMEZ, kayit yazilmaz', async () => {
      const r = await call('POST', '/api/v1/devices/peace-notification/close-all', { token, body: { home_id: homeId, notice_id: noticeId } });
      assert.strictEqual(r.status, 502, JSON.stringify(r.json));
      assert.strictEqual(r.json.code, 'BROKER_UNAVAILABLE');
      assert.strictEqual((await q('SELECT status FROM peace_notification_logs WHERE id = $1', [noticeId])).rows[0].status, 'sent');
      assert.strictEqual((await q("SELECT count(*)::int AS n FROM peace_notification_logs WHERE home_id = $1 AND status = 'manual'", [homeId])).rows[0].n, 0);
    });
    await step('POST close-all gecersiz notice_id -> 400 VALIDATION (yayin yok)', async () => {
      const r = await call('POST', '/api/v1/devices/peace-notification/close-all', { token, body: { home_id: homeId, notice_id: 'abc' } });
      assert.strictEqual(r.status, 400, JSON.stringify(r.json));
      assert.strictEqual(r.json.code, 'VALIDATION');
    });
    await step('DELETE /me/push-tokens -> 200, belirtec devre disi', async () => {
      const r = await call('DELETE', '/api/v1/me/push-tokens', { token, body: { token: fcm } });
      assert.strictEqual(r.status, 200, JSON.stringify(r.json));
      assert.strictEqual(r.json.data.registered, false);
      assert.ok((await q('SELECT disabled_at FROM push_tokens WHERE token = $1', [fcm])).rows[0].disabled_at);
    });

    // ---- zarif kapanis
    const errBefore = errors.length;
    await shutdown('smoke', 0);
    await step('zarif kapanis: cikis kodu 0, kapanis adimlari hatasiz', async () => {
      assert.strictEqual(exitCode, 0);
      const stepErrors = errors.slice(errBefore).filter((e) => /Kapanis adimi basarisiz/.test(e));
      assert.deepStrictEqual(stepErrors, []);
    });
  } catch (e) {
    results.push('FAIL (kurulum/calisma) :: ' + (e && e.stack || e));
  } finally {
    try {
      await q('DELETE FROM push_tokens WHERE user_id = $1', [userId]);
      await q('DELETE FROM device_audit_logs WHERE home_id = $1', [homeId]);
      await q('DELETE FROM homes WHERE id = $1', [homeId]);
      await q('DELETE FROM devices WHERE id = $1', [devId]);
      await q('DELETE FROM users WHERE id = $1', [userId]);
    } catch (e) { results.push('TEMIZLIK HATASI :: ' + e.message); }
    await pool.end().catch(() => {});
    console.error = origErr; console.log = origLog; console.warn = origWarn;
    console.log(results.join('\n'));
    console.log('--- sunucu gunlugu (ilk 25 satir) ---');
    console.log(logs.slice(0, 25).join('\n'));
    console.log('--- console.error (ilk 10) ---');
    console.log(errors.slice(0, 10).join('\n') || '(yok)');
    realExit(results.some((r) => r.startsWith('FAIL') || r.startsWith('TEMIZLIK')) ? 1 : 0);
  }
})();
